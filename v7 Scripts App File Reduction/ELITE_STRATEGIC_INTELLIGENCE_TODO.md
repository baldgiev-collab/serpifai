### Task 17: Create Elite Insight Parser
**File:** `UI/UI_Stage1_Renderer.html`

```javascript
/**
 * V7.22: Parse Elite Strategic Insights from section content
 * @param {string} content - Section markdown content
 * @param {number} sectionNum - Section number (1-14)
 * @returns {Array} Array of insight objects
 */
function parseEliteInsights(content, sectionNum) {
  const insights = [];
  
  // Try JSON block first
  const jsonMatch = content.match(/```json\s*(\{[\s\S]*?"sectionInsights"[\s\S]*?\})\s*```/);
  if (jsonMatch) {
    try {
      const data = JSON.parse(jsonMatch[1]);
      if (data.sectionInsights?.insights) {
        return data.sectionInsights.insights;
      }
    } catch (e) { console.warn('JSON parse failed, using regex'); }
  }
  
  // Fallback: Parse markdown insight blocks
  const insightRegex = /ðŸŽ¯\s*INSIGHT\s*#(\d+):\s*\[?([^\]\n]+)\]?\s*\n-\s*Finding:\s*([^\n]+)\n-\s*(?:Competitive Edge|First-Mover|Competitive Risk):\s*([^\n]+)\n-\s*Actionable Impact:\s*([^\n]+)/gi;
  
  let match;
  while ((match = insightRegex.exec(content)) !== null) {
    insights.push({
      id: parseInt(match[1]),
      type: match[2].trim(),
      finding: match[3].trim(),
      edge: match[4].trim(),
      impact: match[5].trim()
    });
  }
  
  return insights.slice(0, 3);
}
```

```javascript
function buildPillarHierarchy(pillars, centerLabel) {
  const nodes = [];
  const links = [];
  
  // Center node
  nodes.push({
    id: 'center',
    label: centerLabel || 'Content Strategy',
    type: 'center',
    radius: 50,
    color: '#6366f1',
    x: 0, y: 0, fx: 0, fy: 0
  });
  
  // Pillar nodes (ring 1)
  pillars.forEach((pillar, pIdx) => {
    const pillarId = `pillar-${pIdx + 1}`;
    const angle = (2 * Math.PI * pIdx) / pillars.length;
    
    nodes.push({
      id: pillarId,
      label: pillar.name,
      type: 'pillar',
      radius: 35,
      color: getPillarColor(pIdx),
      moatScore: pillar.moatScore,
      clusterCount: pillar.clusters?.length || 0
    });
    
    links.push({ source: 'center', target: pillarId, strength: 0.8 });
    
    // Cluster nodes (ring 2)
    (pillar.clusters || []).forEach((cluster, cIdx) => {
      const clusterId = `cluster-${pIdx + 1}-${cIdx + 1}`;
      
      nodes.push({
        id: clusterId,
        label: cluster.name,
        type: 'cluster',
        radius: 20,
        color: getPillarColor(pIdx, 0.7),
        intent: cluster.intent,
        keywordCount: cluster.keywords?.length || 0
      });
      
      links.push({ source: pillarId, target: clusterId, strength: 0.5 });
      
      // Keyword nodes (ring 3) - only show top 3 for visual clarity
      (cluster.keywords || []).slice(0, 3).forEach((kw, kIdx) => {
        const kwId = `kw-${pIdx + 1}-${cIdx + 1}-${kIdx + 1}`;
        const kwText = typeof kw === 'string' ? kw : (kw.keyword || kw.kw);
        
        nodes.push({
          id: kwId,
          label: kwText,
          type: 'keyword',
          radius: 8,
          color: getPillarColor(pIdx, 0.4),
          volume: kw.vol || kw.volume,
          difficulty: kw.kd || kw.difficulty
        });
        
        links.push({ source: clusterId, target: kwId, strength: 0.3 });
      });
    });
  });
  
  return { nodes, links };
}
```

### Task 33: Create Force Simulation

```javascript
function createForceSimulation(data, width, height) {
  return d3.forceSimulation(data.nodes)
    .force('link', d3.forceLink(data.links)
      .id(d => d.id)
      .distance(d => {
        if (d.source.type === 'center') return 150;
        if (d.source.type === 'pillar') return 80;
        return 40;
      })
      .strength(d => d.strength)
    )
    .force('charge', d3.forceManyBody()
      .strength(d => {
        if (d.type === 'center') return -300;
        if (d.type === 'pillar') return -150;
        if (d.type === 'cluster') return -80;
        return -30;
      })
    )
    .force('center', d3.forceCenter(width / 2, height / 2))
    .force('collision', d3.forceCollide().radius(d => d.radius + 5))
    .alphaDecay(0.02);
}
```

```javascript
function addNodeInteractions(svg, data) {
  svg.selectAll('.node')
    .on('click', function(event, d) {
      event.stopPropagation();
      
      // Highlight connected nodes
      const connectedIds = new Set([d.id]);
      data.links.forEach(link => {
        if (link.source.id === d.id) connectedIds.add(link.target.id);
        if (link.target.id === d.id) connectedIds.add(link.source.id);
      });
      
      svg.selectAll('.node')
        .classed('dimmed', n => !connectedIds.has(n.id))
        .classed('highlighted', n => connectedIds.has(n.id));
      
      // Show tooltip
      showNodeTooltip(event, d);
    })
    .on('mouseover', function(event, d) {
      d3.select(this).select('circle')
        .transition()
        .duration(200)
        .attr('r', d.radius * 1.2);
    })
    .on('mouseout', function(event, d) {
      d3.select(this).select('circle')
        .transition()
        .duration(200)
        .attr('r', d.radius);
    });
  
  // Click background to reset
  svg.on('click', () => {
    svg.selectAll('.node')
      .classed('dimmed', false)
      .classed('highlighted', false);
    hideNodeTooltip();
  });
}
```
