### Task T006: Add CommonCrawl API Integration

**New function in FT_BacklinkExtractor.gs:**

```javascript
/**
 * Fetch backlinks from CommonCrawl Index
 * @param {string} domain - Target domain
 * @returns {Object} Backlink data
 */
function fetchCommonCrawlBacklinks(domain) {
  const ccUrl = `https://index.commoncrawl.org/CC-MAIN-2024-51-index?url=*.${domain}&output=json`;
  
  try {
    const response = UrlFetchApp.fetch(ccUrl, {
      muteHttpExceptions: true,
      timeout: 30000
    });
    
    if (response.getResponseCode() === 200) {
      const lines = response.getContentText().split('\n').filter(Boolean);
      const backlinks = lines.map(line => {
        try {
          const data = JSON.parse(line);
          return {
            url: data.url,
            domain: extractDomainFromUrl(data.url),
            timestamp: data.timestamp,
            source: 'commoncrawl'
          };
        } catch (e) {
          return null;
        }
      }).filter(Boolean);
      
      return {
        success: true,
        backlinks: backlinks,
        count: backlinks.length,
        source: 'CommonCrawl'
      };
    }
  } catch (e) {
    console.log(`⚠️ CommonCrawl error: ${e.message}`);
  }
  
  return { success: false, backlinks: [], count: 0 };
}
```

/**
 * Estimate traffic from Tranco rank
 * Tranco is a research-oriented top sites ranking
 * @param {string} domain - Target domain  
 * @returns {Object} Traffic estimate
 */
function estimateTrafficFromTrancoRank(domain) {
  // Tranco rank to monthly traffic mapping (approximate)
  // Based on industry research correlating rank to traffic
  const rankToTraffic = {
    1: 50000000000,     // Google-level
    100: 500000000,     // Major sites
    1000: 50000000,     // Large sites
    10000: 5000000,     // Medium sites
    100000: 500000,     // Small sites
    1000000: 50000      // Micro sites
  };
  
  // Try to get Tranco rank (cached or API)
  const rank = getTrancoRank(domain);
  
  if (rank > 0) {
    // Interpolate traffic based on rank
    const traffic = Math.round(50000000000 / Math.pow(rank, 0.8));
    return {
      success: true,
      estimatedTraffic: traffic,
      trancoRank: rank,
      confidence: 0.75,
      source: 'Tranco'
    };
  }
  
  return { success: false, estimatedTraffic: 0, confidence: 0 };
